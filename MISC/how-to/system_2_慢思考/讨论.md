# [新增] Fractal System 2: 分形深度思考引擎

class SwarmDeepThink:
    """
    基于 Swarm 的分形 ToT (Tree of Thoughts) 引擎。
    适用于 P2P 全对等网络。任何节点都可以通过此工具动态成为"临时指挥官"。
    """
    def __init__(self, reporter, original_user_id):
        self.reporter = reporter
        self.user_id = original_user_id

    async def execute(self, problem: str) -> str:
        """
        执行流程：
        1. 自身降级为 Orchestrator (指挥官)。
        2. 将思维发散为 3 个具体的、原子的执行任务。
        3. 并行分发给 peers，并强制 peers 进入"EXECUTION_ONLY"模式。
        4. 再分发给一个 peer 做收敛裁判。
        """
        print(f"[Swarm DeepThink] 节点正在发起深度思考: {problem[:30]}...")
        
        # --- Step 1: 策略发散 (Strategy Divergence) ---
        # 关键点：Prompt 必须强制接收者"动手做"，而不是"继续想"
        strategies = [
            ("保守验证", "请编写 Python 代码，使用最基础、稳健的标准库来解决/验证此问题。不要依赖不确定的外部包。"),
            ("算法推演", "请编写 Python 代码，通过数学推导或算法逻辑来模拟此问题。侧重逻辑严密性。"),
            ("暴力测试", "请编写 Python 代码，构建测试用例进行暴力求解或边缘情况探测。侧重实证结果。")
        ]
        
        tasks = []
        for name, instruction in strategies:
            # 【防止递归的关键】：在指令中明确要求 "Do not delegate, just execute code."
            # 我们告诉 Worker：你是执行者，不要再当 Leader 了。
            tasks.append(
                f"【任务类型：原子执行】\n"
                f"任务目标：{problem}\n"
                f"执行策略：{name}\n"
                f"具体指令：{instruction}\n"
                f"IMPORTANT: 你现在是 Worker 角色。请直接调用 Python/Bash 工具执行验证，**严禁**再次调用 deep_think 或 dispatch_task。直接返回代码执行结果。"
            )

        if self.reporter:
            try:
                await self.reporter("update_session_state", {
                    "swarm_mode": "fractal_think",
                    "phase": "diverging"
                })
            except: pass

        # --- Step 2: 并行探索 (Parallel Exploration) ---
        # 复用 dispatch_batch_tasks
        raw_results = await dispatch_batch_tasks(
            tasks=tasks,
            common_context=f"【System 2 模式】当前集群正在进行多路验证。你是被选中的执行节点。",
            return_structured=True,
            _status_reporter=self.reporter,
            _original_user_id=self.user_id
        )

        # 收集结果
        candidates = []
        for res in raw_results:
            if res['success']:
                # 简单清洗，去除 Worker 的废话
                content = res['result']
                candidates.append(f"=== 路径: {strategies[res['index']][0]} ===\n{content}\n")

        if not candidates:
            return "【思考中断】所有子节点均未返回有效结果。请检查代码执行环境。"

        # --- Step 3: 收敛与裁判 (Convergence) ---
        candidates_text = "\n".join(candidates)
        
        # 将裁决任务分派给另一个节点（或者自己做，分派出去可以省自己的 Context）
        judge_instruction = (
            "【任务类型：最终裁决】\n"
            "作为首席架构师，请评估以下 3 个并行任务的执行结果。\n"
            "1. 忽略那些报错或逻辑不通的结果。\n"
            "2. 找出通过代码实证过的最佳答案。\n"
            "3. 整合输出最终结论。\n\n"
            f"--- 原始问题 ---\n{problem}\n\n"
            f"--- 执行结果集 ---\n{candidates_text}"
        )

        if self.reporter:
             try:
                await self.reporter("update_session_state", {"phase": "convergence"})
            except: pass

        final_verdict = await dispatch_task(
            task_instruction=judge_instruction,
            context_info="你现在的角色是 Judge (裁判)。只负责汇总，不需要写代码。",
            target_port=None,
            _status_reporter=self.reporter,
            _original_user_id=self.user_id
        )

        return f"[DeepThink 结论]\n{final_verdict}"

# [新增] Swarm DeepThink: 分布式多路验证引擎
# ==========================================

class SwarmDeepThink:
    def __init__(self, status_reporter, original_user_id):
        self.reporter = status_reporter
        self.user_id = original_user_id

    async def execute(self, problem: str) -> str:
        """
        核心逻辑：
        1. 策略生成：将问题拆解为三个具有不同侧重点的验证指令。
        2. 并行采样：利用 dispatch_batch_tasks 调用 3 个 Worker 同时在各自的沙箱中执行。
        3. 结果一致性校验：指派第 4 个 Worker 作为裁判，根据各路代码执行结果给出最终结论。
        """
        # 步骤 1: 构建多样化的探索任务
        # 强制 Worker 进入"原子执行"模式，防止递归调用 deep_think
        strategies = [
            ("代码实证派", "请编写并运行 Python 脚本，通过实际运行结果来验证此问题的答案。侧重数据真实性。"),
            ("逻辑推演派", "请通过严密的逻辑推断或数学计算来解决此问题。侧重算法的正确性。"),
            ("边缘测试派", "请构建多个测试用例（包括极端情况），验证该问题在各种场景下的表现。")
        ]
        
        tasks = []
        for name, hint in strategies:
            tasks.append(
                f"【System 2 执行指令】\n目标：{problem}\n要求：你是{name}。{hint}\n"
                f"注意：你现在处于执行分支，请直接使用工具(Python/Bash)干活，**严禁**再次派发任务。只需返回最终执行结果。"
            )

        if self.reporter:
            await self.reporter("swarm_event", {"sub_type": "deep_think_start", "data": {"problem": problem}})

        # 步骤 2: 并行分发（宽度搜索）
        # 这里复用你现有的 dispatch_batch_tasks
        raw_results = await dispatch_batch_tasks(
            tasks=tasks,
            common_context=f"这是一次深度思考任务，旨在解决：{problem}",
            return_structured=True,
            _status_reporter=self.reporter,
            _original_user_id=self.user_id
        )

        # 整理有效结果
        candidates = []
        for res in raw_results:
            if res['success']:
                candidates.append(f"### 方案 {res['index']+1} 结果:\n{res['result']}\n")

        if not candidates:
            return "深度思考失败：所有分支 Worker 均未返回有效结果。"

        # 步骤 3: 最终裁决（收敛）
        judge_instruction = (
            "你现在是首席审计员。以下是 3 个节点对同一个问题的并行验证结果。\n"
            "请对比它们的一致性。如果结果冲突，优先相信有代码运行日志支持的那一个。\n"
            f"原始问题：{problem}\n\n"
            f"验证结果汇总：\n" + "\n".join(candidates) +
            "\n请给出最终确定的唯一答案。"
        )

        if self.reporter:
            await self.reporter("swarm_event", {"sub_type": "deep_think_converge", "data": {"count": len(candidates)}})

        # 再次利用一个 Worker 进行总结
        final_verdict = await dispatch_task(
            task_instruction=judge_instruction,
            _status_reporter=self.reporter,
            _original_user_id=self.user_id
        )

        return f"[DeepThink 结论]\n{final_verdict}"

4. 深度解答：为什么没有写明显的执行轮次参数？
你的理解非常精准：Worker 节点自己的 ReAct 循环本身就是“沙箱”和“轮次”的实现载体。

在你的 main_web_start_steering.py 中，每个 Agent 都有一个 run_task 循环，这个循环通常限制在 10-20 次迭代。这就是**“深度”**（Depth）。

无需在 deep_think 里写轮次：因为当你通过 dispatch_task 把任务交给 Worker 时，Worker 会启动它自己的整个 ReAct 流程。Worker 内部如果代码运行报错，它会自动在它的 run_task 循环里进行修正。

沙箱隔离：Worker 在它自己的进程和 Context 空间里反复试错。它的报错日志、中间变量、失败的尝试，统统留在了 Worker 自己的内存里。

结果净化：Worker 只有在它自己的循环结束后（或者修好了 Bug 后），才会通过 [SWARM SUCCESS] 把最终干净的结论回传给 Leader。

总结：

Leader 的 deep_think 控制的是 “宽度” (Width) —— 即：我们要同时尝试几种不同的思路？

Worker 的 run_task 控制的是 “深度” (Depth) —— 即：这一条思路如果不通，我要自我修正多少次？

这种设计实现了关注点分离。Leader 不再关心代码怎么修，它只关心：我的集群里有没有人能通过实证告诉我正确答案。这就是典型的 Fractal Swarm（分形蜂群） 治理模式。
