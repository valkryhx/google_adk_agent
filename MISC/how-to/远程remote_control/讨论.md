# [请插入到 imports 之后，app = FastAPI 初始化之前]
from fastapi.security import HTTPBasic, HTTPBasicCredentials
from fastapi import Depends, HTTPException, status, Request
import secrets

# ⚠️ 配置：请在这里设置你的远程访问密码
REMOTE_PASSWORD = "my_secure_password_123" 

security = HTTPBasic(auto_error=False)

def smart_gatekeeper(credentials: HTTPBasicCredentials = Depends(security), request: Request = None):
    """
    智能门卫逻辑 (Smart Gatekeeper):
    1. 检查请求头是否包含 'cf-ray' -> 如果有，说明来自 Cloudflare -> 强制要求密码。
    2. 如果没有 'cf-ray' -> 说明是本地直连 (Localhost) -> 直接放行 (免密)。
    """
    # Cloudflare Tunnel 会自动给请求添加 cf-ray 头
    is_remote = "cf-ray" in request.headers

    # === 本地访问：直接放行 ===
    if not is_remote:
        return "local_user_admin"

    # === 远程访问：必须验证 ===
    if not credentials:
        # 触发浏览器原生的登录弹窗
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Remote Access Requires Login",
            headers={"WWW-Authenticate": "Basic"},
        )

    # 安全地比对密码
    # 用户名可以是任意值（例如 'admin'），我们只校验密码
    is_password_correct = secrets.compare_digest(credentials.password, REMOTE_PASSWORD)
    
    if not is_password_correct:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Incorrect Password",
            headers={"WWW-Authenticate": "Basic"},
        )
    return credentials.username
